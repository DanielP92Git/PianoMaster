```sql
-- === Q1: real column list of parent_subscriptions ===
SELECT a.attnum,
       a.attname                                   AS column_name,
       format_type(a.atttypid, a.atttypmod)        AS data_type,
       a.attnotnull                                AS not_null,
       pg_get_expr(d.adbin, d.adrelid)             AS default_expr
FROM   pg_attribute a
LEFT   JOIN pg_attrdef d ON d.adrelid = a.attrelid AND d.adnum = a.attnum
WHERE  a.attrelid = 'public.parent_subscriptions'::regclass
  AND  a.attnum > 0 AND NOT a.attisdropped
ORDER  BY a.attnum;

-- === Q2: constraints, incl. the existing student_id FK's ON DELETE rule ===
--  confdeltype: a=NO ACTION, r=RESTRICT, c=CASCADE, n=SET NULL, d=SET DEFAULT
SELECT c.conname, c.contype, c.confdeltype,
       pg_get_constraintdef(c.oid) AS definition
FROM   pg_constraint c
WHERE  c.conrelid = 'public.parent_subscriptions'::regclass
ORDER  BY c.contype, c.conname;

-- === Q3: current RLS policies on the table (D-15 additive baseline) ===
SELECT policyname, permissive, roles, cmd, qual, with_check
FROM   pg_policies
WHERE  schemaname = 'public' AND tablename = 'parent_subscriptions'
ORDER  BY policyname;

-- === Q4: current has_active_subscription() definition (D-16 baseline) ===
SELECT pg_get_functiondef('public.has_active_subscription(uuid)'::regprocedure);

-- === Q5: D-11 "before" — every live subscription row, with the parent_id the
--         D-01 resolve-chain WOULD assign. Emails are shown UNREDACTED here for
--         your eyes only; the committed artifact records the redacted form. ===
SELECT ps.ls_subscription_id,
       ps.student_id                                   AS legacy_student_id,
       COALESCE(p.id, cp.parent_id)                    AS would_resolve_to_parent_id,
       CASE WHEN p.id IS NOT NULL      THEN 'probe1_parents'
            WHEN cp.parent_id IS NOT NULL THEN 'probe2_child_profiles'
            ELSE 'UNRESOLVED' END                      AS resolve_probe,
       ps.parent_email                                 AS ls_parent_email,
       au.email                                        AS auth_email_of_resolved_parent,
       ps.status,
       ps.current_period_end
FROM   parent_subscriptions ps
LEFT   JOIN parents        p  ON p.id  = ps.student_id
LEFT   JOIN child_profiles cp ON cp.id = ps.student_id
LEFT   JOIN auth.users     au ON au.id = COALESCE(p.id, cp.parent_id)
ORDER  BY ps.created_at;

-- === Q6: D-08 duplicate-active audit (read-only; NO constraint is being added) ===
SELECT student_id, COUNT(*) AS active_rows
FROM   parent_subscriptions
WHERE  status IN ('active','on_trial')
    OR (status = 'cancelled' AND current_period_end > NOW())
    OR (status = 'past_due'  AND current_period_end > NOW() - INTERVAL '3 days')
GROUP  BY student_id
HAVING COUNT(*) > 1;

-- === Q7: total row count (sanity: SC-2 says 3 live subscriptions) ===
SELECT COUNT(*) AS total_rows FROM parent_subscriptions;
```
