export async function deleteChildProfile(childId, confirmationNickname) {
const { isOwner } = await verifyStudentDataAccess(childId);
if (!isOwner) throw new Error("Unauthorized");
const { data: child } = await supabase.from("child_profiles").select("nickname").eq("id", childId).single();
if (confirmationNickname.toLowerCase().trim() !== child.nickname.toLowerCase()) throw new Error("Name does not match");
const { error } = await supabase.from("students").delete().eq("id", childId); // cascade trigger removes downstream
if (error) throw error;
// NO supabase.auth.signOut() — parent stays logged in (D-12)
}

| --------------------------------- | --------------------------------------------- |
| client-supplied childId → service | export/delete act on a client-provided id     |
| service → Supabase delete         | irreversible cascade delete of a child's data |

| Threat ID  | Category                           | Component                             | Disposition | Mitigation Plan                                                                                                                                                                                                                                   |
| ---------- | ---------------------------------- | ------------------------------------- | ----------- | ------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| T-04-03-01 | Elevation of Privilege             | verifyStudentDataAccess parent branch | mitigate    | Branch verifies true DB ownership via `child_profiles.eq(id).eq(parent_id, user.id)` under RLS; never trusts a client-supplied id. Returns isOwner only on a real row; a non-owned id is rejected — proven by the behavioral owned/non-owned test |
| T-04-03-02 | Tampering / Repudiation            | deleteChildProfile                    | mitigate    | Guarded by (a) verifyStudentDataAccess ownership and (b) case-insensitive nickname confirmation before the irreversible `students.delete()`; RLS additionally blocks non-owned rows                                                               |
