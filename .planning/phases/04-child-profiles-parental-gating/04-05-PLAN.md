ParentGateContext (in-memory only — NEVER localStorage/sessionStorage):
  const WINDOW_MS = 3 * 60 * 1000;   // Claude's discretion 2-5 min → 3 min
  // The gated (parent-only) surfaces are /parent-portal, /manage-children, /settings, /subscribe, /legal.
  // Every OTHER post-auth surface is a child surface; navigating to one closes the shared window (D-05 trigger 4).
  const CHILD_SURFACE_ROUTES = ["/", "/dashboard", "/trail", "/notes-master-mode",
    "/rhythm-mode", "/ear-training-mode", "/practice-modes", "/practice-sessions",
    "/achievements", "/assignments"];
  const isChildSurfaceRoute = (pathname) =>
    pathname === "/" || CHILD_SURFACE_ROUTES.some((r) => r !== "/" && pathname.startsWith(r));
  state: passedAt (number|null, in a useRef or useState but NOT persisted)
  passed  = passedAt !== null && Date.now() - passedAt < WINDOW_MS
  pass()      -> setPassedAt(Date.now()); (re)arm a setTimeout(closeGate, WINDOW_MS)   // trigger 1: timeout
  closeGate() -> setPassedAt(null); clear timer
  effects: window.addEventListener("visibilitychange", () => document.hidden && closeGate())  // trigger 3
           window.addEventListener("blur", closeGate)                                          // trigger 3
           on ActiveChild switch → closeGate  (subscribe via useActiveChild(); close when activeChildId changes)  // trigger 2
           const location = useLocation();
           useEffect(() => { if (isChildSurfaceRoute(location.pathname)) closeGate(); }, [location.pathname])      // trigger 4
  exports: ParentGateProvider, useParentGate  (co-located hook triad like SettingsContext)

ParentGateProtectedRoute (mirror src/ui/ProtectedRoute.jsx — full 31 lines):
function ParentGateProtectedRoute({ children }) {
const { passed, pass } = useParentGate();
const navigate = useNavigate();
if (!passed) return <ParentGateMath onConsent={pass} onCancel={() => navigate(-1)} isRTL={isRTL} />;
return children;
}
// The component remounts per route entry, so the !passed check is inherently a mount re-check (no URL bypass).
